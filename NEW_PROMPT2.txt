## 5.1 NAME VALIDATION

```python
import re

def validate_name(value, field_name="Name"):
    if not isinstance(value, str) or not value.strip():
        return f"{field_name} is required."

    value = value.strip()

    if len(value) > 100:
        return f"{field_name} is too long."

    parts = re.split(r"[ '-]", value)

    if not all(
        part and all(character.isalpha() for character in part)
        for part in parts
    ):
        return (
            f"{field_name} may contain letters, spaces, "
            "apostrophes, and hyphens only."
        )

    return None
```

## 5.2 USERNAME VALIDATION

```python
def validate_username(username):
    if not isinstance(username, str):
        return "Username is required."

    if not re.fullmatch(
        r"[A-Za-z0-9_.-]{3,30}",
        username.strip()
    ):
        return (
            "Username must contain 3–30 characters using "
            "letters, numbers, periods, underscores, or hyphens."
        )

    return None
```

Only use this if the existing system has usernames.

## 5.3 PASSWORD VALIDATION

```python
def validate_password(password, confirmation=None):
    if not isinstance(password, str):
        return "Password is required."

    if len(password) < 8:
        return "Password must contain at least 8 characters."

    if not re.search(r"[A-Z]", password):
        return "Password must contain an uppercase letter."

    if not re.search(r"[a-z]", password):
        return "Password must contain a lowercase letter."

    if not re.search(r"[0-9]", password):
        return "Password must contain a number."

    if confirmation is not None and password != confirmation:
        return "Passwords do not match."

    return None
```

Use this for new passwords, not to reject existing passwords during login.

## 5.4 PHILIPPINE MOBILE NUMBER VALIDATION

```python
def validate_phone(phone):
    if not isinstance(phone, str):
        return "Phone number is required."

    phone = phone.strip()

    if not re.fullmatch(r"09\d{9}", phone):
        return (
            "Enter an 11-digit Philippine mobile "
            "number starting with 09."
        )

    if re.search(r"(\d)\1{4}", phone):
        return (
            "Phone number cannot contain five "
            "identical consecutive digits."
        )

    return None
```

## 5.5 PASSENGER COUNT VALIDATION

```python
def validate_passengers(value, boat_capacity):
    if isinstance(value, bool):
        return "Passenger count must be a whole number."

    text = str(value).strip()

    if not re.fullmatch(r"[0-9]+", text):
        return "Passenger count must be a whole number."

    passengers = int(text)

    if passengers < 1:
        return "At least one passenger is required."

    if passengers > boat_capacity:
        return "Passenger count exceeds boat capacity."

    return None
```
